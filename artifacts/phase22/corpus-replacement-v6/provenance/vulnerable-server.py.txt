36: class BearerTokenAuthMiddleware:
37:     """ASGI middleware that requires one configured bearer token."""
38: 
39:     def __init__(self, app: Any, token: str) -> None:
40:         self.app = app
41:         self.expected_authorization = f"Bearer {token}"
42: 
43:     async def __call__(self, scope: dict[str, Any], receive: Any, send: Any) -> None:
44:         if scope.get("type") != "http":
45:             await self.app(scope, receive, send)
46:             return
47: 
48:         authorization = _authorization_header(scope)
49:         if authorization is None or not compare_digest(
50:             authorization,
51:             self.expected_authorization,
52:         ):
53:             await _send_unauthorized(send)
54:             return
55: 
56:         await self.app(scope, receive, send)
57: 
58: 
59: def _authorization_header(scope: dict[str, Any]) -> str | None:
60:     for name, value in scope.get("headers", []):
61:         if name.lower() == b"authorization":
62:             return value.decode("latin-1")
63:     return None
64: 
65: 
66: async def _send_unauthorized(send: Any) -> None:
67:     body = b"Unauthorized"
68:     await send(
69:         {
70:             "type": "http.response.start",
71:             "status": 401,
72:             "headers": [
73:                 (b"content-type", b"text/plain; charset=utf-8"),
74:                 (b"www-authenticate", b"Bearer"),
75:                 (b"content-length", str(len(body)).encode("ascii")),
76:             ],
77:         }
78:     )
79:     await send({"type": "http.response.body", "body": body})

82: def _csv_env(name: str, default: list[str]) -> list[str]:
83:     raw_value = os.getenv(name, "")
84:     values = [value.strip() for value in raw_value.split(",") if value.strip()]
85:     return values or default
86: 
87: 
88: def _has_wildcard(values: list[str]) -> bool:
89:     return "*" in values
90: 
91: 
92: def _http_auth_token() -> str | None:
93:     token = os.getenv("MCP_AUTH_TOKEN")
94:     if token is None or not token.strip():
95:         return None
96:     return token.strip()
97: 
98: 
99: def _build_transport_security_settings(
100:     allowed_hosts: list[str],
101:     allowed_origins: list[str],
102: ) -> TransportSecuritySettings:
103:     wildcard_requested = _has_wildcard(allowed_hosts) or _has_wildcard(allowed_origins)
104:     return TransportSecuritySettings(
105:         enable_dns_rebinding_protection=not wildcard_requested,
106:         allowed_hosts=allowed_hosts,
107:         allowed_origins=allowed_origins,
108:     )
109: 
110: 
111: def _configure_http_settings() -> None:
112:     host = os.getenv("MCP_HOST", "0.0.0.0")
113:     port = int(os.getenv("MCP_PORT", "8000"))
114:     allowed_hosts = _csv_env("MCP_ALLOWED_HOSTS", ["*"])
115:     allowed_origins = _csv_env("MCP_ALLOWED_ORIGINS", ["*"])
116: 
117:     mcp.settings.host = host
118:     mcp.settings.port = port
119:     mcp.settings.transport_security = _build_transport_security_settings(
120:         allowed_hosts=allowed_hosts,
121:         allowed_origins=allowed_origins,
122:     )
123: 
124: 
125: async def _run_http_async() -> None:
126:     import uvicorn
127: 
128:     auth_token = _http_auth_token()
129:     if auth_token is None:
130:         LOGGER.warning(
131:             "MCP HTTP transport is running without MCP_AUTH_TOKEN; "
132:             "all HTTP requests are unauthenticated."
133:         )
134: 
135:     starlette_app = mcp.streamable_http_app()
136:     if auth_token is not None:
137:         starlette_app.add_middleware(BearerTokenAuthMiddleware, token=auth_token)
138: 
139:     config = uvicorn.Config(
140:         starlette_app,
141:         host=mcp.settings.host,
142:         port=mcp.settings.port,
143:         log_level=mcp.settings.log_level.lower(),
144:     )
145:     server = uvicorn.Server(config)
146:     await server.serve()

311: def _prune_cache_on_startup() -> None:
312:     try:
313:         stats = cache_ops.prune_cache_on_startup()
314:     except Exception:  # noqa: BLE001 - cache pruning must not block server startup.
315:         LOGGER.warning("Startup cache pruning failed", exc_info=True)
316:         return
317:     if stats is not None and stats.deleted_files:
318:         LOGGER.info(
319:             "Pruned %s cache files on startup (%s bytes)",
320:             stats.deleted_files,
321:             stats.deleted_bytes,
322:         )
