37: class KnownPlaceholderTokenError(RuntimeError):
38:     """Raised when MCP_AUTH_TOKEN is a publicly known placeholder."""
39: 
40: 
41: KNOWN_PLACEHOLDER_TOKENS = {"change-me-now", "changeme", "secret", "token"}
42: 
43: 
44: class BearerTokenAuthMiddleware:
45:     """ASGI middleware that requires one configured bearer token."""
46: 
47:     def __init__(self, app: Any, token: str) -> None:
48:         self.app = app
49:         self.expected_authorization = f"Bearer {token}"
50: 
51:     async def __call__(self, scope: dict[str, Any], receive: Any, send: Any) -> None:
52:         scope_type = scope.get("type")
53:         if scope_type == "lifespan":
54:             await self.app(scope, receive, send)
55:             return
56:         if scope_type != "http":
57:             # Fail closed: unknown or websocket scopes never bypass the token.
58:             raise RuntimeError(
59:                 f"BearerTokenAuthMiddleware refuses to serve scope type {scope_type!r}"
60:             )
61: 
62:         authorization = _authorization_header(scope)
63:         if authorization is None or not _safe_token_match(
64:             authorization, self.expected_authorization
65:         ):
66:             await _send_unauthorized(send)
67:             return
68: 
69:         await self.app(scope, receive, send)
70: 
71: 
72: def _safe_token_match(left: str, right: str) -> bool:
73:     try:
74:         return compare_digest(left.encode("utf-8"), right.encode("utf-8"))
75:     except (TypeError, UnicodeEncodeError):
76:         return False
77: 
78: 
79: def _authorization_header(scope: dict[str, Any]) -> str | None:
80:     for name, value in scope.get("headers", []):
81:         if name.lower() == b"authorization":
82:             return value.decode("latin-1")
83:     return None
84: 
85: 
86: async def _send_unauthorized(send: Any) -> None:
87:     body = b"Unauthorized"
88:     await send(
89:         {
90:             "type": "http.response.start",
91:             "status": 401,
92:             "headers": [
93:                 (b"content-type", b"text/plain; charset=utf-8"),
94:                 (b"www-authenticate", b"Bearer"),
95:                 (b"content-length", str(len(body)).encode("ascii")),
96:             ],
97:         }
98:     )
99:     await send({"type": "http.response.body", "body": body})

102: def _csv_env(name: str, default: list[str]) -> list[str]:
103:     raw_value = os.getenv(name, "")
104:     values = [value.strip() for value in raw_value.split(",") if value.strip()]
105:     return values or default
106: 
107: 
108: def _has_wildcard(values: list[str]) -> bool:
109:     return "*" in values
110: 
111: 
112: def _http_auth_token() -> str | None:
113:     token = os.getenv("MCP_AUTH_TOKEN")
114:     if token is None or not token.strip():
115:         return None
116:     return token.strip()
117: 
118: 
119: def _build_transport_security_settings(
120:     allowed_hosts: list[str],
121:     allowed_origins: list[str],
122: ) -> TransportSecuritySettings:
123:     wildcard_requested = _has_wildcard(allowed_hosts) or _has_wildcard(allowed_origins)
124:     return TransportSecuritySettings(
125:         enable_dns_rebinding_protection=not wildcard_requested,
126:         allowed_hosts=allowed_hosts,
127:         allowed_origins=allowed_origins,
128:     )
129: 
130: 
131: def _configure_http_settings() -> None:
132:     host = os.getenv("MCP_HOST", "0.0.0.0")
133:     raw_port = os.getenv("MCP_PORT", "8000")
134:     try:
135:         port = int(raw_port)
136:     except ValueError as exc:
137:         raise RuntimeError(
138:             f"Invalid MCP_PORT={raw_port!r}; expected an integer port number"
139:         ) from exc
140:     if not 1 <= port <= 65535:
141:         raise RuntimeError(f"Invalid MCP_PORT={raw_port!r}; out of range")
142:     allowed_hosts = _csv_env("MCP_ALLOWED_HOSTS", ["*"])
143:     allowed_origins = _csv_env("MCP_ALLOWED_ORIGINS", ["*"])
144: 
145:     mcp.settings.host = host
146:     mcp.settings.port = port
147:     mcp.settings.transport_security = _build_transport_security_settings(
148:         allowed_hosts=allowed_hosts,
149:         allowed_origins=allowed_origins,
150:     )
151: 
152: 
153: async def _run_http_async() -> None:
154:     import uvicorn
155: 
156:     auth_token = _http_auth_token()
157:     if auth_token is None:
158:         if not _get_env_bool("MCP_ALLOW_UNAUTHENTICATED_HTTP", False):
159:             raise RuntimeError(
160:                 "MCP HTTP transport requires MCP_AUTH_TOKEN to be set. "
161:                 "Set a strong random token, or set MCP_ALLOW_UNAUTHENTICATED_HTTP=1 "
162:                 "to explicitly run without authentication."
163:             )
164:         LOGGER.warning(
165:             "MCP HTTP transport is running without MCP_AUTH_TOKEN; "
166:             "all HTTP requests are unauthenticated."
167:         )
168:     elif auth_token.lower() in KNOWN_PLACEHOLDER_TOKENS:
169:         raise KnownPlaceholderTokenError(
170:             f"MCP_AUTH_TOKEN is set to the publicly known placeholder {auth_token!r}. "
171:             "Set a strong random token before exposing the server over HTTP."
172:         )
173: 
174:     starlette_app = mcp.streamable_http_app()
175:     if auth_token is not None:
176:         starlette_app.add_middleware(BearerTokenAuthMiddleware, token=auth_token)
177: 
178:     config = uvicorn.Config(
179:         starlette_app,
180:         host=mcp.settings.host,
181:         port=mcp.settings.port,
182:         log_level=mcp.settings.log_level.lower(),
183:     )
184:     server = uvicorn.Server(config)
185:     await server.serve()

350: def _prune_cache_on_startup() -> None:
351:     try:
352:         stats = cache_ops.prune_cache_on_startup()
353:     except Exception:  # noqa: BLE001 - cache pruning must not block server startup.
354:         LOGGER.warning("Startup cache pruning failed", exc_info=True)
355:         return
356:     if stats is not None and stats.deleted_files:
357:         LOGGER.info(
358:             "Pruned %s cache files on startup (%s bytes)",

364: def main() -> None:
365:     _prune_cache_on_startup()

367:     if transport in {"http", "streamable-http"}:
368:         import asyncio
369: 
370:         _configure_http_settings()
371:         asyncio.run(_run_http_async())
372:         return
373:     if transport != "stdio":
374:         raise RuntimeError(
375:             f"Invalid MCP_TRANSPORT={transport!r}; accepted values are 'stdio' and 'http'"
376:         )
