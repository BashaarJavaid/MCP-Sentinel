57: class StrictBaseModel(BaseModel):
58:     """Base model that rejects unexpected tool arguments."""
59: 
60:     model_config = ConfigDict(extra="forbid")
61: 
62: 
63: class SearchRequest(StrictBaseModel):
64:     query: str = Field(min_length=1, description=ARGUMENT_DESCRIPTIONS["query"])
65:     max_results: int = Field(
66:         default=10,
67:         ge=1,
68:         le=30,
69:         description=ARGUMENT_DESCRIPTIONS["max_results"],
70:     )
71:     search_window: int | None = Field(
72:         default=None,
73:         ge=1,
74:         le=100,
75:         description=ARGUMENT_DESCRIPTIONS["search_window"],
76:     )
77:     safe_search: SafeSearch = Field(default="off", description=ARGUMENT_DESCRIPTIONS["safe_search"])
78:     time_filter: TimeFilter | None = Field(
79:         default=None,
80:         description=ARGUMENT_DESCRIPTIONS["time_filter"],
81:     )
82:     blocked_domains: list[str] = Field(
83:         default_factory=list,
84:         description=ARGUMENT_DESCRIPTIONS["blocked_domains"],
85:     )
86:     allowed_domains: list[str] = Field(
87:         default_factory=list,
88:         description=ARGUMENT_DESCRIPTIONS["allowed_domains"],
89:     )
90:     preferred_domains: list[str] = Field(
91:         default_factory=list,
92:         description=ARGUMENT_DESCRIPTIONS["preferred_domains"],
93:     )
94: 
95:     @field_validator("blocked_domains", "allowed_domains", "preferred_domains")
96:     @classmethod
97:     def normalize_domain_options(cls, values: list[str]) -> list[str]:
98:         return normalize_domains(values)
